        const parts = str.split(' ');
        return parts.map((part, i) => {
          if (part.includes('/')) {
            const [num, den] = part.split('/');
            return <Fraction key={i} num={num} den={den} />;
          } else {
            return <span key={i} className="mx-2 font-bold">{part}</span>;
          }
        });
      };

      // --- メインアプリ ---
      function App() {
        const [gameState, setGameState] = useState('start');
        const [currentLevel, setCurrentLevel] = useState('初級');
        const [questionIndex, setQuestionIndex] = useState(0);
        const [score, setScore] = useState(0);
        const [isCorrect, setIsCorrect] = useState(false);
        const [shuffledOptions, setShuffledOptions] = useState([]);

        const currentQ = questionsData[currentLevel][questionIndex];

        useEffect(() => {
          if (gameState === 'playing' && currentQ) {
            setShuffledOptions([...currentQ.options].sort(() => Math.random() - 0.5));
          }
        }, [gameState, questionIndex, currentLevel]);

        const handleStart = () => {
          setCurrentLevel('初級');
          setQuestionIndex(0);
          setScore(0);
          setGameState('playing');
        };

        const handleAnswer = (option) => {
          const correct = option === currentQ.ans;
          setIsCorrect(correct);
          if (correct) setScore(prev => prev + 1);
          setGameState('explanation');
        };

        const handleNextQuestion = () => {
          if (questionIndex < 9) {
            setQuestionIndex(prev => prev + 1);
            setGameState('playing');
          } else {
            setGameState('level_result');
          }
        };

        const handleNextLevel = () => {
          const levelIdx = levels.indexOf(currentLevel);
          if (levelIdx < levels.length - 1) {
            setCurrentLevel(levels[levelIdx + 1]);
            setQuestionIndex(0);
            setScore(0);
            setGameState('playing');
          } else {
            setGameState('all_clear');
          }
        };

                        </div>
                        <button onClick={handleNextLevel} className="w-full py-4 bg-pink-500 hover:bg-pink-600 text-white rounded-full text-xl font-bold shadow-lg transition-transform active:scale-95 flex items-center justify-center gap-2">
                          <span className="mr-1">{currentLevel === '上級' ? 'エンディングへ！' : '次の級へ進む！'}</span> <Icons.ArrowRight />
                        </button>
                      </React.Fragment>
                    ) : (
                      <React.Fragment>
                        <div className="text-8xl mb-6 opacity-80">🐰💦</div>
                        <div className="bg-blue-50 border-2 border-blue-200 p-6 rounded-3xl relative mb-8 w-full">
                          <div className="absolute w-4 h-4 bg-blue-50 border-t-2 border-l-2 border-blue-200 rotate-45 -top-[9px] left-1/2 -translate-x-1/2"></div>
                          <p className="text-xl font-bold text-blue-800 leading-relaxed">合格まであと少し…！<br/>8問以上の正解を目指して<br/>もう一度チャレンジしよう！</p>
                        </div>
                        <button onClick={handleRetry} className="w-full py-4 bg-blue-500 hover:bg-blue-600 text-white rounded-full text-xl font-bold shadow-lg transition-transform active:scale-95 flex items-center justify-center gap-2">
                          <Icons.RotateCcw /> もう一度挑戦する
                        </button>
                      </React.Fragment>
                    )}
                  </div>
                )}

                {/* 全クリ画面 */}
                {gameState === 'all_clear' && (
                  <div className="text-center flex flex-col items-center justify-center h-full">
                    <div className="inline-block bg-yellow-100 text-yellow-600 font-bold px-4 py-1 rounded-full mb-4">Congratulations!</div>
                    <h2 className="text-4xl font-extrabold text-yellow-500 mb-8">全級クリア！！</h2>
                    <div className="text-9xl mb-8 relative">
                      <span className="absolute -top-10 left-1/2 -translate-x-1/2 text-6xl drop-shadow-md">👑</span>
                      <span className="inline-block" style={{ animation: 'bounce 1.5s infinite' }}>🐰</span>
                    </div>
                    <div className="bg-yellow-50 p-6 rounded-3xl mb-10 w-full border-2 border-yellow-200">
                      <p className="text-lg text-yellow-900 font-bold leading-relaxed">
                        おめでとうございます！🎉<br/><br/>
                        分数の加減と乗除の計算方法、<br/>もうバッチリだね！<br/>
                        通分が必要なとき、逆数にするとき、<br/>
                        しっかり区別できるようになったよ！
                      </p>
                    </div>
                    <button onClick={handleStart} className="w-full py-4 bg-gradient-to-r from-pink-400 to-pink-500 hover:from-pink-500 hover:to-pink-600 text-white rounded-full text-xl font-bold shadow-lg transition-transform active:scale-95 flex items-center justify-center gap-2">
                      <Icons.RotateCcw /> 最初から遊ぶ
                    </button>
                  </div>
                )}
              </div>
              
              {/* プログレスバー */}
              {gameState === 'playing' && (
                <div className="h-2 bg-pink-100 w-full">
                  <div className="h-full bg-pink-400 transition-all duration-300 ease-out" style={{ width: `${(questionIndex / 10) * 100}%` }}></div>
                </div>
              )}
            </div>
          </div>
        );
      }

      // Reactアプリの描画を開始
      const root = ReactDOM.createRoot(document.getElementById('root'));
      root.render(<App />);
    </script>
  </body>
